cmake_minimum_required(VERSION 3.20)
project(lab_device_variant12 VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ENABLE_COVERAGE "Instrument the model and recycle detector with GCC coverage" OFF)
option(ENABLE_SANITIZERS "Enable address/undefined behavior sanitizers on Linux" OFF)

add_library(device_model device.cpp recycle.cpp)
target_include_directories(device_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(device_model PRIVATE /W4 /WX)
else()
  target_compile_options(device_model PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(ENABLE_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "Coverage requires GCC and its matching gcov")
  endif()
  target_compile_options(device_model PRIVATE -O0 -g --coverage)
  target_link_options(device_model PUBLIC --coverage)
endif()
if(ENABLE_SANITIZERS)
  if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers are supported with GCC/Clang on Unix")
  endif()
  target_compile_options(device_model PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(device_model PUBLIC -fsanitize=address,undefined)
endif()

add_executable(lab_demo main.cpp)
target_link_libraries(lab_demo PRIVATE device_model)

include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.zip
    URL_HASH SHA256=40d4ec942217dcc84a9ebe2a68584ada7d4a33a8ee958755763278ea1c5e18ff
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
  )
  FetchContent_MakeAvailable(googletest)
  add_executable(lab_tests tests/test_main.cpp tests/test_device.cpp tests/test_recycle.cpp)
  target_link_libraries(lab_tests PRIVATE device_model GTest::gtest)
  if(NOT MSVC)
    target_compile_options(lab_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  include(GoogleTest)
  gtest_discover_tests(lab_tests DISCOVERY_TIMEOUT 30)
endif()
